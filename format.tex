% Equation numbers restart inside every problem and subproblem and carry the
% problem label, so they stay unique document-wide: (3.1), (3.a.1), (3.b.1), ...
\newcommand{\resetequationnumbering}[1]{%
  \setcounter{equation}{0}%
  \def\theequation{#1.\arabic{equation}}%
  \def\theHequation{#1.\arabic{equation}}%
}

% Notes drawn on a problem that runs past a page boundary. They sit just
% outside the frame so they never collide with the box content. They name the
% problem, never the part: a broken box draws its overlays only after its whole
% content is set, so which part straddles the break is not knowable here.
\newcommand{\problemcontinues}[1]{%
  \draw[black!35, dashed, line width = 0.3pt]
  (frame.south west) -- (frame.south east);% chktex 8
  \node[anchor = north east, font = \scriptsize\itshape, text = black!60,
    inner sep = 0pt] at ([yshift = -4pt] frame.south east)
  {Problem #1 continues on next page};%
}
\newcommand{\problemcontinued}[1]{%
  \draw[black!35, dashed, line width = 0.3pt]
  (frame.north west) -- (frame.north east);% chktex 8
  \node[anchor = south west, font = \scriptsize\itshape, text = black!60,
    inner sep = 0pt] at ([yshift = 3pt] frame.north west)
  {Problem #1 continued from previous page};%
}

\newtcbtheorem[]{problembox}{Problem}%
{enhanced,
  before upper = {\resetequationnumbering{\thetcb@cnt@problembox}},
  % Set per segment, not via `overlay first and middle` plus
  % `overlay middle and last`: those both \let the middle overlay, so the
  % second one silently wins and middle pages lose a note.
  overlay first = {\problemcontinues{\thetcb@cnt@problembox}},
  overlay middle = {\problemcontinued{\thetcb@cnt@problembox}%
    \problemcontinues{\thetcb@cnt@problembox}},
  overlay last = {\problemcontinued{\thetcb@cnt@problembox}},
  colback = black!5, %white,
  colbacktitle = black!5,
  coltitle = black,
  boxrule = 0pt,
  frame hidden,
  borderline west = {0.5mm}{0.0mm}{black},
  fonttitle = \bfseries\sffamily,
  breakable,
  before skip = 3ex,
  after skip = 3ex
}{problem}

% Both arguments are optional: \begin{problem}, \begin{problem}[Title], or
% \begin{problem}[Title][label]. Use [][label] for a label with no title.
%
% \newtcbtheorem insists on two mandatory arguments, so it generates
% `problembox` and this wraps it. The box cannot simply be \let to another name
% and redefined in place: the environment it generates is a LaTeX3 document
% command that resolves its own argument spec by name, so the alias still
% dispatches to whatever `problem` currently means, and the wrapper recurses
% until the input stack overflows.
\NewDocumentEnvironment{problem}{O{}O{}}
{\begin{problembox}{#1}{#2}}
{\end{problembox}}

% Parts are lettered within their problem: Problem 3.a, Problem 3.b, ...
%
% A plain environment, deliberately not a nested tcbtheorem. A breakable
% tcolorbox inside a breakable tcolorbox behaves atomically: when the part does
% not fit it is moved whole to the next page, which stranded about two inches at
% the bottom of pages. `enforce breakable` does split it, but in absurd places
% (measured on this assignment: 9 pages, or 16 with \allowdisplaybreaks, against
% 6). With one box level the content flows normally and \allowdisplaybreaks can
% finally take effect, because the part is no longer an unsplittable unit.
\newcounter{subproblem}[tcb@cnt@problembox]
\renewcommand{\thesubproblem}{\thetcb@cnt@problembox.\alph{subproblem}}
\NewDocumentEnvironment{subproblem}{O{}O{}}{%
  \par\addvspace{2ex}%
  \refstepcounter{subproblem}%
  % Second optional argument is the label:
  % \begin{subproblem}[][foo] is referenced as \cref{subproblem:foo}.
  \if\relax\detokenize{#2}\relax\else\label{subproblem:#2}\fi
  \resetequationnumbering{\thesubproblem}%
  \leftskip=1.5em\relax
  \noindent{\bfseries\sffamily Problem \thesubproblem%
    \if\relax\detokenize{#1}\relax\else: #1\fi}%
  % Keep the heading with at least the start of its part.
  \par\nobreak\vspace{0.5ex}\nobreak%
}{%
  \par\addvspace{2ex}%
}

% cleveref keys names off the counter name. `problem` is still a tcbtheorem,
% and tcolorbox names its counter after the generated environment, so that
% is the name to use; getting it wrong leaves \cref printing `??` in front of an
% otherwise correct number. `subproblem` is a plain environment with its own
% counter, so it is named directly.
%
% Label either one with its second optional argument rather than \label:
% \begin{subproblem}[][foo] is referenced as \cref{subproblem:foo}. For
% `problem` a \label inside the body records an empty number, because tcolorbox
% does not set \@currentlabel there; `subproblem` uses \refstepcounter and so
% does not have that restriction.
\crefname{tcb@cnt@problembox}{Problem}{Problems}
\crefname{subproblem}{Problem}{Problems}
\Crefname{tcb@cnt@problembox}{Problem}{Problems}
\Crefname{subproblem}{Problem}{Problems}
\makeatother
